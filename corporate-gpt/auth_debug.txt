"""
Authentication Routes — FastAPI Users
Uses FastAPI Users library for all auth operations.
Provides thin custom endpoints on top of FastAPI Users defaults.
"""
from fastapi import APIRouter, Depends
from fastapi_users import models as fu_models
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.providers.auth import (
    fastapi_users,
    auth_backend,
    current_active_user,
    current_superuser,
)
from app.core.database import get_db
from app.models.user import User

router = APIRouter(prefix="/auth", tags=["auth"])

# ── FastAPI Users built-in routes ──
router.include_router(
    fastapi_users.get_auth_router(auth_backend),
    prefix="",
)
router.include_router(
    fastapi_users.get_register_router(),
    prefix="",
)
router.include_router(
    fastapi_users.get_verify_router(),
    prefix="",
)
router.include_router(
    fastapi_users.get_reset_password_router(),
    prefix="",
)

# ── Custom routes ──

@router.get("/me", response_model=fu_models.UserRead)
async def get_current_user_info(
    user: User = Depends(current_active_user),
):
    """Get current user information."""
    return user


@router.get("/users")
async def list_users(
    user: User = Depends(current_superuser),
    db: AsyncSession = Depends(get_db),
):
    """List all users (super admin only)."""
    result = await db.execute(select(User))
    users = result.scalars().all()
    return users
